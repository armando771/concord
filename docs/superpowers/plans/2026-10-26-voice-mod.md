# Voice mod e refatoração da Mesa de Áudio — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cada usuário escolhe um preset de voz (com sliders de tom, timbre e intensidade) dentro da Mesa de Áudio, e todos na chamada ouvem a voz alterada.

**Architecture:** O efeito entra na cadeia Web Audio do microfone (`noiseSuppression.ts`), depois do ganho de entrada e antes da detecção de voz. Um AudioWorklet faz o pitch shift granular (e a modulação em anel do "Robô"); filtros e delays nativos fazem timbre, rádio e eco; a lógica de presets é pura e testável. A Mesa de Áudio vira um contêiner com trilho lateral (Sons · Voz).

**Tech Stack:** TypeScript, React, zustand (`persist`), Web Audio / AudioWorklet, vitest.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-10-26-voice-mod-design.md`.
- Presets da v1, exatamente estes: **Normal, Grave, Esquilo, Robô, Eco, Rádio**. Sem voz que imite pessoa real.

// frontend/src/renderer/src/lib/voiceModPresets.ts
// Voice mod: presets e a tradução deles para parâmetros de áudio.
//
// Puro (sem DOM nem Web Audio): é usado pelo store, pelo grafo (voiceMod.ts) e
// nos testes.

export type VoiceModPreset = 'normal' | 'grave' | 'esquilo' | 'robo' | 'eco' | 'radio'
/** Efeito fixo de cada preset (os sliders ajustam por cima). */
export type VoiceModEffect = 'none' | 'ring' | 'echo' | 'radio'

export interface VoiceModValues {
  /** Semitons, −12…+12. */
  pitch: number
  /** −1 (voz mais grossa) … +1 (mais fina). */
  timbre: number
  /** 0 = voz original … 1 = só a voz processada. */
  intensity: number
}

export interface VoiceModSettings extends VoiceModValues {
  /** Preset de base: define o efeito e os valores iniciais dos sliders. */
  preset: VoiceModPreset
  /** Algum slider foi mexido depois de escolher o preset. */
  customized: boolean
}

export const PITCH_MIN = -12
export const PITCH_MAX = 12

export const PRESET_ORDER: VoiceModPreset[] = ['normal', 'grave', 'esquilo', 'robo', 'eco', 'radio']

export const VOICE_MOD_PRESETS: Record<
  VoiceModPreset,
  { label: string; effect: VoiceModEffect; values: VoiceModValues }
> = {
  normal: { label: 'Normal', effect: 'none', values: { pitch: 0, timbre: 0, intensity: 1 } },
  grave: { label: 'Grave', effect: 'none', values: { pitch: -5, timbre: -0.5, intensity: 1 } },
  esquilo: { label: 'Esquilo', effect: 'none', values: { pitch: 7, timbre: 0.5, intensity: 1 } },
  robo: { label: 'Robô', effect: 'ring', values: { pitch: 0, timbre: 0.2, intensity: 1 } },
  eco: { label: 'Eco', effect: 'echo', values: { pitch: 0, timbre: 0, intensity: 0.6 } },
  radio: { label: 'Rádio', effect: 'radio', values: { pitch: 0, timbre: 0.3, intensity: 1 } }
}

export function clampPitch(pitch: number): number {
  return Math.min(PITCH_MAX, Math.max(PITCH_MIN, pitch))
}

export function clampTimbre(timbre: number): number {
  return Math.min(1, Math.max(-1, timbre))
}

export function clampIntensity(intensity: number): number {
  return Math.min(1, Math.max(0, intensity))
}

/** Frequência do oscilador do "Robô" (modulação em anel). */
export const RING_HZ = 50
/** Quanto o timbre empurra os shelves de graves e agudos, em dB. */
export const TIMBRE_SHELF_DB = 8

export interface VoiceModParams {
  /** Sem tom, timbre nem efeito: o sinal passa direto, sem latência extra. */
  bypass: boolean
  pitchRatio: number
  /** 0 = sem modulação em anel. */
  ringHz: number
  echo: boolean
  radio: boolean
  lowShelfDb: number
  highShelfDb: number
  dry: number
  wet: number
}

export function voiceModParams(settings: VoiceModSettings): VoiceModParams {
  const { effect } = VOICE_MOD_PRESETS[settings.preset]
  return {
    bypass: effect === 'none' && settings.pitch === 0 && settings.timbre === 0,
    pitchRatio: 2 ** (settings.pitch / 12),
    ringHz: effect === 'ring' ? RING_HZ : 0,
    echo: effect === 'echo',
    radio: effect === 'radio',
    lowShelfDb: -settings.timbre * TIMBRE_SHELF_DB,
    highShelfDb: settings.timbre * TIMBRE_SHELF_DB,
    dry: 1 - settings.intensity,
    wet: settings.intensity
  }
}

export function isVoiceModActive(settings: VoiceModSettings): boolean {
  return !voiceModParams(settings).bypass
}
```

// frontend/src/renderer/src/store/voiceModStore.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import {
  clampIntensity,
  clampPitch,
  clampTimbre,
  VOICE_MOD_PRESETS,
  type VoiceModPreset,
  type VoiceModSettings
} from '../lib/voiceModPresets'

// Voice mod (Mesa de Áudio → Voz), guardado neste PC. O efeito é aplicado no
// microfone de quem fala, então todos na chamada ouvem a voz alterada.

interface VoiceModState extends VoiceModSettings {
  /** O worklet carregou no último microfone aberto (falso = voice mod indisponível). */
  available: boolean
  /** Ouvindo a própria voz alterada (não persiste). */
  monitoring: boolean
  selectPreset: (preset: VoiceModPreset) => void
  setPitch: (pitch: number) => void
  setTimbre: (timbre: number) => void
  setIntensity: (intensity: number) => void
  setAvailable: (available: boolean) => void
  setMonitoring: (monitoring: boolean) => void
}

export const useVoiceModStore = create<VoiceModState>()(
  persist(
    (set) => ({
      preset: 'normal',
      customized: false,
      ...VOICE_MOD_PRESETS.normal.values,
      available: true,
      monitoring: false,
      selectPreset: (preset) =>
        set({ preset, customized: false, ...VOICE_MOD_PRESETS[preset].values }),
      setPitch: (pitch) => set({ pitch: clampPitch(pitch), customized: true }),
      setTimbre: (timbre) => set({ timbre: clampTimbre(timbre), customized: true }),
      setIntensity: (intensity) => set({ intensity: clampIntensity(intensity), customized: true }),
      setAvailable: (available) => set({ available }),
      setMonitoring: (monitoring) => set({ monitoring })
    }),
    {
      name: 'concord-voice-mod',
      partialize: ({ preset, customized, pitch, timbre, intensity }) => ({
        preset,
        customized,
        pitch,
        timbre,
        intensity
      })
    }
  )
)

// frontend/src/renderer/src/lib/voiceModDsp.test.ts
import { describe, expect, it } from 'vitest'
import { VOICE_MOD_LATENCY_MS, VoiceModDsp } from './voiceModDsp'

const RATE = 48_000
const BLOCK = 128

/** 1 s de sinal (375 blocos de 128 amostras). */
function signal(fn: (i: number) => number): Float32Array {
  return Float32Array.from({ length: RATE }, (_, i) => fn(i))
}

function sine(hz: number): Float32Array {
  return signal((i) => 0.5 * Math.sin((2 * Math.PI * hz * i) / RATE))
}

function run(dsp: VoiceModDsp, input: Float32Array): Float32Array {
  const out = new Float32Array(input.length)
  for (let i = 0; i < input.length; i += BLOCK) {
    dsp.process(input.subarray(i, i + BLOCK), out.subarray(i, i + BLOCK))
  }
  return out
}

/** Frequência estimada pelas passagens por zero, ignorando o primeiro quarto de segundo. */
function frequency(out: Float32Array): number {
  const from = Math.round(RATE * 0.25)
  let crossings = 0
  for (let i = from + 1; i < out.length; i++) if (out[i - 1] < 0 && out[i] >= 0) crossings++
  return crossings / ((out.length - from) / RATE)
}

describe('VoiceModDsp', () => {
  it('razão 1 devolve a entrada atrasada em 25 ms', () => {
    const delay = Math.round((RATE * VOICE_MOD_LATENCY_MS) / 1000)
    const input = sine(440)
    const out = run(new VoiceModDsp(RATE), input)
    for (let i = delay + 10; i < out.length; i += 97) {
      expect(out[i]).toBeCloseTo(input[i - delay], 4)
    }
  })

  it('razão 2 sobe uma oitava', () => {
    const dsp = new VoiceModDsp(RATE)
    dsp.ratio = 2
    expect(frequency(run(dsp, sine(1000)))).toBeCloseTo(2000, -2)
  })

  it('razão 0,5 desce uma oitava', () => {
    const dsp = new VoiceModDsp(RATE)
    dsp.ratio = 0.5
    expect(frequency(run(dsp, sine(1000)))).toBeCloseTo(500, -2)
  })

  it('a amplitude se mantém (as pontas somam 1)', () => {
    const dsp = new VoiceModDsp(RATE)
    dsp.ratio = 1.5
    const out = run(dsp, sine(1000))
    let peak = 0
    for (let i = RATE / 2; i < out.length; i++) peak = Math.max(peak, Math.abs(out[i]))
    expect(peak).toBeGreaterThan(0.45)
    expect(peak).toBeLessThan(0.55)
  })

  it('modulação em anel multiplica a voz por um seno', () => {
    const dsp = new VoiceModDsp(RATE)
    dsp.ringHz = 100
    expect(frequency(run(dsp, signal(() => 1)))).toBeCloseTo(100, -1)

  it('sem entrada conta como silêncio', () => {
    const dsp = new VoiceModDsp(RATE)
    const out = new Float32Array(BLOCK)
    dsp.process(undefined, out)
    expect(out.every((v) => v === 0)).toBe(true)
  })
})

// frontend/src/renderer/src/lib/voiceModDsp.ts
// Núcleo do voice mod: pitch shift granular e modulação em anel.
//
// Auto-contido (sem DOM): roda dentro do AudioWorklet (voiceModWorklet.ts) e
// nos testes.
//
// Pitch shift: buffer circular com duas pontas de leitura defasadas em meia
// janela. O atraso de cada ponta varre a janela a (1 − razão) amostras por
// amostra, então a leitura avança `razão` amostras por amostra. As pontas se
// alternam com janelas sin² que somam 1 (sem tremor de volume). Com razão 1 a
// saída é a entrada atrasada em meia janela.

/** Nome do processador registrado pelo voiceModWorklet. */
export const VOICE_MOD_PROCESSOR = 'concord-voice-mod'

/** Janela do pitch shift. */
const WINDOW_MS = 50
/** Atraso fixo do caminho processado (meia janela); o caminho seco compensa. */
export const VOICE_MOD_LATENCY_MS = WINDOW_MS / 2

export interface VoiceModWorkletMessage {
  /** Razão de frequência (2 = uma oitava acima). */
  ratio: number
  /** Frequência da modulação em anel; 0 = desligada. */
  ringHz: number
}

function sinSquared(phase: number): number {
  const s = Math.sin(Math.PI * phase)
  return s * s
}

export class VoiceModDsp {
  ratio = 1
  ringHz = 0
  private readonly rate: number
  private readonly size: number
  private readonly buffer: Float32Array
  private pos = 0
  private phase = 0
  private ringPhase = 0

  constructor(rate: number) {
    this.rate = rate
    this.size = Math.round((rate * WINDOW_MS) / 1000)
    this.buffer = new Float32Array(this.size * 2 + 4)
  }

  process(input: Float32Array | undefined, output: Float32Array): void {
    const ringStep = (2 * Math.PI * this.ringHz) / this.rate
    for (let i = 0; i < output.length; i++) {
      this.buffer[this.pos] = input?.[i] ?? 0
      const phaseB = (this.phase + 0.5) % 1
      let y =
        this.read(this.phase * this.size) * sinSquared(this.phase) +
        this.read(phaseB * this.size) * sinSquared(phaseB)
      if (this.ringHz > 0) {
        y *= Math.sin(this.ringPhase)
        this.ringPhase = (this.ringPhase + ringStep) % (2 * Math.PI)
      }
      output[i] = y
      this.phase += (1 - this.ratio) / this.size
      this.phase -= Math.floor(this.phase)
      this.pos = (this.pos + 1) % this.buffer.length
    }
  }

  /** Lê o buffer `delay` amostras atrás da última escrita (interpolação linear). */
  private read(delay: number): number {
    const n = this.buffer.length
    const p = this.pos - delay
    const i0 = Math.floor(p)
    const frac = p - i0
    const a = this.buffer[((i0 % n) + n) % n]
    const b = this.buffer[(((i0 + 1) % n) + n) % n]
    return a + (b - a) * frac
  }
}
```

  readonly port: MessagePort
}

  private dsp = new VoiceModDsp(sampleRate)

  constructor() {

      this.dsp.ringHz = event.data.ringHz
    }

    const output = outputs[0]?.[0]
    if (output) this.dsp.process(inputs[0]?.[0], output)
    return true
  }
}

// frontend/src/renderer/src/lib/voiceMod.ts
// Grafo Web Audio do voice mod. Vive dentro do AudioContext do microfone
// (noiseSuppression.ts), depois do ganho de entrada.
//
//   in ─ bypass ───────────────────────────────────────────────┐
//   in ─ dryDelay ─ dry ───────────────────────────────────────┤
//   in ─ shifter ─ rádio ─ timbre ─ wet ─┬─────────────────────┴─ out
//                              └─ eco ───┘
//
// Rádio e eco ficam sempre no grafo e são neutralizados pelos parâmetros
// (passa-faixa aberto, curva nula, ganho do eco em 0): trocar de preset só
// mexe em valores, nunca reconecta nós, então vale ao vivo sem estalo.
// O caminho seco tem o mesmo atraso do pitch shifter para não haver filtro
// pente ao misturar os dois.
import voiceModWorkletUrl from './voiceModWorklet.ts?worker&url'
import {
  VOICE_MOD_LATENCY_MS,
  VOICE_MOD_PROCESSOR,
  type VoiceModWorkletMessage
} from './voiceModDsp'
import { voiceModParams, type VoiceModSettings } from './voiceModPresets'

/** Constante de tempo do crossfade ao trocar de preset ou mexer num slider. */
const FADE_SECONDS = 0.015
const ECHO_DELAY_SECONDS = 0.28
const ECHO_FEEDBACK = 0.35
const ECHO_LEVEL = 0.6
const RADIO_HIGHPASS_HZ = 300
const RADIO_LOWPASS_HZ = 3400
/** Valores "abertos" dos filtros do rádio: passam o sinal sem mexer. */
const OPEN_HIGHPASS_HZ = 10
const OPEN_LOWPASS_HZ = 20_000

/** Saturação suave do rádio. */
const RADIO_CURVE = (() => {
  const curve = new Float32Array(1024)
  for (let i = 0; i < curve.length; i++) {
    const x = (i / (curve.length - 1)) * 2 - 1
    curve[i] = Math.tanh(3 * x) / Math.tanh(3)
  }
  return curve
})()

export interface VoiceModNode {
  input: AudioNode
  output: AudioNode
  /** `instant`: sem rampa (primeira aplicação, antes de o áudio rodar). */
  apply: (settings: VoiceModSettings, instant: boolean) => void
  destroy: () => void
}

/** Rejeita se o worklet não carregar; quem chama segue sem voice mod. */
export async function createVoiceMod(ctx: AudioContext): Promise<VoiceModNode> {
  await ctx.audioWorklet.addModule(voiceModWorkletUrl)

  const input = ctx.createGain()
  input.channelCount = 1
  input.channelCountMode = 'explicit'
  const output = ctx.createGain()
  const bypass = ctx.createGain()
  const dry = ctx.createGain()
  const wet = ctx.createGain()
  const dryDelay = ctx.createDelay(0.1)
  dryDelay.delayTime.value = VOICE_MOD_LATENCY_MS / 1000
  const shifter = new AudioWorkletNode(ctx, VOICE_MOD_PROCESSOR, {
    numberOfInputs: 1,
    numberOfOutputs: 1,
    outputChannelCount: [1]
  })
  const radioHigh = ctx.createBiquadFilter()
  radioHigh.type = 'highpass'
  const radioLow = ctx.createBiquadFilter()
  radioLow.type = 'lowpass'
  const radioClip = ctx.createWaveShaper()
  const low = ctx.createBiquadFilter()
  low.type = 'lowshelf'
  low.frequency.value = 250
  const high = ctx.createBiquadFilter()
  high.type = 'highshelf'
  high.frequency.value = 3000
  const echoDelay = ctx.createDelay(1)
  echoDelay.delayTime.value = ECHO_DELAY_SECONDS
  const echoFeedback = ctx.createGain()
  echoFeedback.gain.value = ECHO_FEEDBACK
  const echoLevel = ctx.createGain()

  input.connect(bypass).connect(output)
  input.connect(dryDelay).connect(dry).connect(output)
  input
    .connect(shifter)
    .connect(radioHigh)
    .connect(radioLow)
    .connect(radioClip)
    .connect(low)
    .connect(high)
    .connect(wet)
    .connect(output)
  high.connect(echoDelay).connect(echoFeedback).connect(echoDelay)
  echoDelay.connect(echoLevel).connect(wet)

  const nodes: AudioNode[] = [
    input, output, bypass, dry, wet, dryDelay, shifter, radioHigh, radioLow, radioClip,
    low, high, echoDelay, echoFeedback, echoLevel
  ]

  const set = (param: AudioParam, value: number, instant: boolean): void => {
    const t = ctx.currentTime
    param.cancelScheduledValues(t)
    if (instant) param.setValueAtTime(value, t)
    else param.setTargetAtTime(value, t, FADE_SECONDS)
  }

  return {
    input,
    output,
    apply: (settings, instant) => {
      const p = voiceModParams(settings)
      set(bypass.gain, p.bypass ? 1 : 0, instant)
      set(dry.gain, p.bypass ? 0 : p.dry, instant)
      set(wet.gain, p.bypass ? 0 : p.wet, instant)
      set(low.gain, p.lowShelfDb, instant)
      set(high.gain, p.highShelfDb, instant)
      set(radioHigh.frequency, p.radio ? RADIO_HIGHPASS_HZ : OPEN_HIGHPASS_HZ, instant)
      set(radioLow.frequency, p.radio ? RADIO_LOWPASS_HZ : OPEN_LOWPASS_HZ, instant)
      set(echoLevel.gain, p.echo ? ECHO_LEVEL : 0, instant)
      radioClip.curve = p.radio ? RADIO_CURVE : null
      const message: VoiceModWorkletMessage = { ratio: p.pitchRatio, ringHz: p.ringHz }
      shifter.port.postMessage(message)
    },
    destroy: () => nodes.forEach((node) => node.disconnect())
  }
}

function SoundboardPopover({
  anchor,
  onClose
}: {
  anchor: HTMLElement
  onClose: () => void
}): React.JSX.Element | null {
  const ref = useRef<HTMLDivElement>(null)
  const serverId = useCallStore((s) => s.serverId)
  const voiceModActive = useVoiceModStore(isVoiceModActive)
  const [section, setSection] = useState<Section>('sounds')
  // Modal de upload ou gravação de atalho aberto na seção Sons: não fechar por fora.
  const [busy, setBusy] = useState(false)
  const handleBusy = useCallback((next: boolean) => setBusy(next), [])

  useEffect(() => {
    const onDown = (e: MouseEvent): void => {
      if (busy) return
      const target = e.target as Node
      if (!ref.current?.contains(target) && !anchor.contains(target)) {
        onClose()
      }
    }
    const onKey = (e: KeyboardEvent): void => {
      if (busy) return
      if (e.key === 'Escape') onClose()
    }
    document.addEventListener('mousedown', onDown)
    document.addEventListener('keydown', onKey)
    return () => {
      document.removeEventListener('mousedown', onDown)
      document.removeEventListener('keydown', onKey)
    }
  }, [anchor, onClose, busy])

  if (!serverId) return null

  const rect = anchor.getBoundingClientRect()
  const left = Math.min(
    Math.max(MARGIN, rect.left + rect.width / 2 - POPOVER_WIDTH / 2),
    window.innerWidth - POPOVER_WIDTH - MARGIN
  )
  const bottom = Math.max(MARGIN, window.innerHeight - rect.top + GAP)
  const current = SECTIONS.find((s) => s.id === section) ?? SECTIONS[0]

  return createPortal(
    <div
      ref={ref}
      role="dialog"
      aria-label="Mesa de Áudio"
      style={{ width: POPOVER_WIDTH, left, bottom }}
      className="fixed z-50 flex max-h-[460px] gap-3 rounded-2xl border border-white/10 bg-[#111214] p-3 shadow-2xl"
    >
      <div role="tablist" aria-orientation="vertical" className="flex shrink-0 flex-col gap-1.5 border-r border-white/8 pr-3">
        {SECTIONS.map(({ id, label }) => {
          const selected = id === section
          return (
            <button
              key={id}
              type="button"
              role="tab"
              aria-selected={selected}
              aria-label={label}
              title={label}
              onClick={() => setSection(id)}
              className={`relative flex h-10 w-10 cursor-pointer items-center justify-center rounded-xl transition-colors ${
                selected
                  ? 'bg-[#5865f2] text-white'
                  : 'bg-white/[0.06] text-gray-300 hover:bg-white/[0.12] hover:text-white'
              }`}
            >
              {id === 'sounds' ? (
                <SoundPadsIcon className="h-5 w-5" />
              ) : (
                <MicIcon className="h-5 w-5" />
              )}
              {id === 'voice' && voiceModActive && (
                <span
                  aria-hidden="true"
                  className="absolute -right-0.5 -top-0.5 h-2.5 w-2.5 rounded-full border-2 border-[#111214] bg-[#23a55a]"
                />
              )}
            </button>
          )
        })}
      </div>

      <div className="flex min-h-0 min-w-0 flex-1 flex-col">
        <div className="mb-3">
          <p className="text-sm font-bold text-white">Mesa de Áudio · {current.label}</p>
          <p className="text-[11px] text-gray-400">{current.subtitle}</p>
        </div>
        {section === 'sounds' ? (
          <SoundboardSoundsPanel serverId={serverId} onBusyChange={handleBusy} />
        ) : (
          <VoiceModPanel />
        )}
      </div>
    </div>,

  )
}
